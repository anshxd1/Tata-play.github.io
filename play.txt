function configureShakaPlayer(mpdUrl) {
  shakaPlayer.configure({
    drm: { clearKeys: streamConfig.drm },
    streaming: {
      bufferingGoal: 30,
      rebufferingGoal: 15,
      retryParameters: { maxAttempts: 5 }
    }
  });

  shakaPlayer.getNetworkingEngine().clearAllRequestFilters();
  shakaPlayer.getNetworkingEngine().registerRequestFilter((type, request) => {
    request.headers['Referer'] = streamConfig.referer;
    request.headers['User-Agent'] = streamConfig.userAgent;
    if ((type === shaka.net.NetworkingEngine.RequestType.MANIFEST ||
         type === shaka.net.NetworkingEngine.RequestType.SEGMENT) &&
        !request.uris[0].includes('__hdnea')) {
      const sep = request.uris[0].includes('?') ? '&' : '?';
      request.uris[0] += sep + streamConfig.token;
    }
  });

  shakaPlayer.load(mpdUrl).then(() => {
    console.log('Stream loaded:', mpdUrl);
  }).catch(e => console.error('Load error', e));
}

// ===== CATCHUP MODAL =====
function openCatchup() {
  const modal = document.getElementById('catchupModal');

  // Set max/min constraints
  const nowIST = getNowIST();
  const sevenDaysAgoIST = new Date(nowIST.getTime() - 7 * 24 * 60 * 60 * 1000);

  const startInput = document.getElementById('catchupStart');
  const endInput = document.getElementById('catchupEnd');

  startInput.min = toLocalInputValue(sevenDaysAgoIST);
  startInput.max = toLocalInputValue(nowIST);
  endInput.max = toLocalInputValue(nowIST);

  // Default: last 1 hour
  const oneHourAgo = new Date(nowIST.getTime() - 60 * 60 * 1000);
  startInput.value = toLocalInputValue(oneHourAgo);
  endInput.value = toLocalInputValue(nowIST);

  document.getElementById('catchupError').style.display = 'none';
  modal.classList.add('open');
}

function closeCatchup() {
  document.getElementById('catchupModal').classList.remove('open');
}

}

// Get current time as if in IST (we use local browser time + offset to IST)
function getNowIST() {
  // IST = UTC+5:30
  const now = new Date();
  const utcMs = now.getTime() + now.getTimezoneOffset() * 60000;
  return new Date(utcMs + 5.5 * 60 * 60 * 1000);
}

// Format date for datetime-local input: "YYYY-MM-DDTHH:MM"
function toLocalInputValue(date) {
  const pad = n => String(n).padStart(2, '0');
  return `${date.getFullYear()}-${pad(date.getMonth()+1)}-${pad(date.getDate())}T${pad(date.getHours())}:${pad(date.getMinutes())}`;
}

// Convert IST datetime-local string → UTC timestamp string for MPD (YYYYMMDDTHHmmss)
function istInputToUTCParam(istStr) {
  // istStr is "YYYY-MM-DDTHH:MM" in IST (UTC+5:30)
  const [datePart, timePart] = istStr.split('T');
  const [year, month, day] = datePart.split('-').map(Number);
  const [hour, minute] = timePart.split(':').map(Number);

  // Build IST date, then subtract 5h30m to get UTC
  const istMs = Date.UTC(year, month - 1, day, hour, minute, 0) - (5 * 60 + 30) * 60 * 1000;
  const utcDate = new Date(istMs);

  const pad = n => String(n).padStart(2, '0');
  return `${utcDate.getUTCFullYear()}${pad(utcDate.getUTCMonth()+1)}${pad(utcDate.getUTCDate())}T${pad(utcDate.getUTCHours())}${pad(utcDate.getUTCMinutes())}00`;
}

function validateCatchup() {
  const errEl = document.getElementById('catchupError');
  const startVal = document.getElementById('catchupStart').value;
  const endVal = document.getElementById('catchupEnd').value;

  if (!startVal || !endVal) { errEl.style.display = 'none'; return true; }

  const nowIST = getNowIST();
  const sevenDaysAgo = new Date(nowIST.getTime() - 7 * 24 * 60 * 60 * 1000);

  const startDate = new Date(startVal.replace('T', ' '));
  const endDate = new Date(endVal.replace('T', ' '));

  if (startDate < sevenDaysAgo) {
    showCatchupError('Start time cannot be more than 7 days ago.');
    return false;
  }
  if (endDate > nowIST) {
    showCatchupError('End time cannot be in the future.');
    return false;
  }
  if (startDate >= endDate) {
    showCatchupError('Start time must be before end time.');
    return false;
  }

  errEl.style.display = 'none';
  return true;
}

function showCatchupError(msg) {
  const errEl = document.getElementById('catchupError');
  errEl.textContent = msg;
  errEl.style.display = 'block';
}

function playCatchup() {
  if (!validateCatchup()) return;

  const startVal = document.getElementById('catchupStart').value;
  const endVal = document.getElementById('catchupEnd').value;

  if (!startVal || !endVal) {
    showCatchupError('Please select both start and end times.');
    return;
  }

  const beginParam = istInputToUTCParam(startVal);
  const endParam = istInputToUTCParam(endVal);

  // Append ?begin=...&end=... to the base MPD URL
  const baseMpd = streamConfig.mpd.split('?')[0]; // strip existing params
  const catchupMpd = `${baseMpd}?begin=${beginParam}&end=${endParam}`;

  closeCatchup();
  showToast('Loading catchup stream…');

  // Reload shaka with catchup MPD
  shakaPlayer.unload().then(() => {
    configureShakaPlayer(catchupMpd);
  });
}

// ===== EPG PROGRESS =====
function updateProgress() {
  const parse = t => {
    const [h, m] = t.split(':');
    const d = new Date();
    d.setHours(Number(h), Number(m), 0, 0);
    return d;
  };
  const start = parse("13:00:00");
  const end = parse("14:00:00");
  const now = new Date();
  if (end < start) end.setDate(end.getDate() + 1);
  const pct = Math.max(0, Math.min(100, ((now - start) / (end - start)) * 100));
  document.getElementById('epg-progress').style.width = pct + '%';
}
setInterval(updateProgress, 5000);
updateProgress();

  }
}
